from typing import Dict, Any
import asyncio

class BaseAgent(ABC):
    def __init__(self, message_bus, agent_id: str):
        self.message_bus = message_bus
        self.agent_id = agent_id
        self.state = {}
        self.running = False
    
    @abstractmethod
    async def process(self, data: Dict[str, Any]) -> Dict[str, Any]:
        pass
    
    async def start(self):
        self.running = True
        await self.message_bus.subscribe(f"{self.agent_id}_input", self.handle_message)
    
    async def stop(self):
        self.running = False
    
    async def publish(self, channel: str, data: Dict[str, Any]):
        await self.message_bus.publish(channel, data)

What is an Embedding Model?
Think of an embedding model as a translator that converts human language into mathematical coordinates that computers can understand and work with.

The Simple Explanation
Imagine you have a magical translator that can take any piece of text and convert it into a specific location in a multi-dimensional space - kind of like GPS coordinates, but instead of just X and Y, it might use 384, 768, or even 1536 dimensions.

Example:

"I love pizza" → [0.2, -0.1, 0.8, 0.3, ...] (384 numbers)
"Pizza is delicious" → [0.18, -0.09, 0.79, 0.31, ...] (384 numbers)
"I hate vegetables" → [-0.3, 0.4, -0.2, 0.1, ...] (384 numbers)
Notice how the first two are very similar (close coordinates) because they express similar ideas, while the third is different.

How It Works Technically
Training Process
Feed massive amounts of text to a neural network
Teach it to predict what words come next
Extract the internal representations the network learned
These representations become embeddings
The Magic
The network learns that words with similar meanings should have similar mathematical representations. It discovers patterns like:

King - Man + Woman ≈ Queen
Paris - France + Germany ≈ Berlin

Copyfrom sentence_transformers import SentenceTransformer
import numpy as np
from typing import List, Union
import asyncio

class EmbeddingService:
    def __init__(self, model_name: str = "all-MiniLM-L6-v2"):
        self.model = SentenceTransformer(model_name)
        self.embedding_cache = {}
        self.dimension = self.model.get_sentence_embedding_dimension()
    
    async def embed_text(self, text: str) -> np.ndarray:
        """Convert single text to embedding vector"""
        if text in self.embedding_cache:
            return self.embedding_cache[text]
        
        # Run in thread pool to avoid blocking
        loop = asyncio.get_event_loop()
        embedding = await loop.run_in_executor(
            None, self.model.encode, text
        )
        
        # Cache result
        self.embedding_cache[text] = embedding
        return embedding
    
    async def embed_batch(self, texts: List[str]) -> List[np.ndarray]:
        """Convert multiple texts to embeddings efficiently"""
        loop = asyncio.get_event_loop()
        embeddings = await loop.run_in_executor(
            None, self.model.encode, texts
        )
        return embeddings
    
    def similarity(self, embedding1: np.ndarray, embedding2: np.ndarray) -> float:
        """Calculate cosine similarity between embeddings"""
        return np.dot(embedding1, embedding2) / (
            np.linalg.norm(embedding1) * np.linalg.norm(embedding2)
        )
    
    async def find_most_similar(self, query: str, candidates: List[str], top_k: int = 5):
        """Find most similar texts to query"""
        query_embedding = await self.embed_text(query)
        candidate_embeddings = await self.embed_batch(candidates)
        
        similarities = [
            self.similarity(query_embedding, candidate_embedding)
            for candidate_embedding in candidate_embeddings
        ]
        
        # Get top-k results
        top_indices = np.argsort(similarities)[-top_k:][::-1]
        results = [
            {"text": candidates[i], "similarity": similarities[i]}
            for i in top_indices
        ]
        
        return results

        }

Smart Routing for Maximum Speed
Lightning-Fast Model Selection:
Copyclass FastModelRouter:
    def __init__(self):
        # Pre-compiled regex patterns for instant routing
        self.patterns = {
            "commands": re.compile(r'\b(turn|start|stop|open|close|set|play|pause)\b'),
            "questions": re.compile(r'\b(what|how|when|where|why|which|who)\b'),
            "technical": re.compile(r'\b(script|code|file|config|api|error|debug)\b'),
        }
        
    def quick_route(self, text: str) -> str:
        """Ultra-fast routing in <5ms"""
        text_lower = text.lower()
        
        # Pattern matching is much faster than LLM classification
        for model_type, pattern in self.patterns.items():
            if pattern.search(text_lower):
                return model_type
        
        return "general"  # Default fallback
        
    # This is 100x faster than using an LLM to classify intent!
